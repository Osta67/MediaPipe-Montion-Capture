import streamlit as st import cv2 import mediapipe as mp import numpy as np import time # --- PAGE CONFIG --- st.set_page_config(page_title="FitRise AI", layout="wide") # --- CUSTOM STYLING --- st.markdown(""" """, unsafe_allow_html=True) # --- TITLE --- st.markdown("
🏋️ FitRise AI – Smart Squat Trainer
", unsafe_allow_html=True) st.write("") # --- LAYOUT --- col1, col2 = st.columns([3, 1]) with col2: rep_box = st.empty() stage_box = st.empty() angle_box = st.empty() progress_bar = st.progress(0) with col1: frame_placeholder = st.empty() # --- MEDIAPIPE SETUP --- mp_pose = mp.solutions.pose mp_drawing = mp.solutions.drawing_utils # --- STATE INIT --- if "reps" not in st.session_state: st.session_state.reps = 0 if "stage" not in st.session_state: st.session_state.stage = "up" # --- ANGLE FUNCTION --- def calculate_angle(a, b, c): a, b, c = np.array(a), np.array(b), np.array(c) radians = np.arctan2(c[1]-b[1], c[0]-b[0]) - np.arctan2(a[1]-b[1], a[0]-b[0]) angle = np.abs(radians * 180.0 / np.pi) return 360 - angle if angle > 180 else angle # --- CAMERA --- cap = cv2.VideoCapture(0) # --- MAIN LOOP --- with mp_pose.Pose(min_detection_confidence=0.7, min_tracking_confidence=0.7) as pose: while cap.isOpened() and st.session_state.reps < 10: ret, frame = cap.read() if not ret: st.error("❌ Camera not detected") break # --- PREPROCESS --- frame = cv2.flip(frame, 1) image = cv2.cvtColor(frame, cv2.COLOR_BGR2RGB) image.flags.writeable = False results = pose.process(image) image.flags.writeable = True image = cv2.cvtColor(image, cv2.COLOR_RGB2BGR) angle = 0 try: landmarks = results.pose_landmarks.landmark hip = [landmarks[mp_pose.PoseLandmark.RIGHT_HIP.value].x, landmarks[mp_pose.PoseLandmark.RIGHT_HIP.value].y] knee = [landmarks[mp_pose.PoseLandmark.RIGHT_KNEE.value].x, landmarks[mp_pose.PoseLandmark.RIGHT_KNEE.value].y] ankle = [landmarks[mp_pose.PoseLandmark.RIGHT_ANKLE.value].x, landmarks[mp_pose.PoseLandmark.RIGHT_ANKLE.value].y] angle = calculate_angle(hip, knee, ankle) # --- REP LOGIC --- if angle > 165: st.session_state.stage = "up" if angle < 95 and st.session_state.stage == "up": st.session_state.stage = "down" st.session_state.reps += 1 # --- DRAW ANGLE --- h, w, _ = image.shape knee_pos = tuple(np.multiply(knee, [w, h]).astype(int)) cv2.putText(image, f"{int(angle)}°", knee_pos, cv2.FONT_HERSHEY_SIMPLEX, 0.6, (0, 255, 136), 2, cv2.LINE_AA) except: pass # --- DRAW LANDMARKS --- mp_drawing.draw_landmarks( image, results.pose_landmarks, mp_pose.POSE_CONNECTIONS, mp_drawing.DrawingSpec(color=(0,255,136), thickness=2, circle_radius=2), mp_drawing.DrawingSpec(color=(255,255,255), thickness=2) ) # --- UI UPDATE --- frame_placeholder.image(image, channels="BGR", use_container_width=True) rep_box.markdown(f"""

Reps

{st.session_state.reps}

""", unsafe_allow_html=True) stage_class = "status-up" if st.session_state.stage == "up" else "status-down" stage_box.markdown(f"""

Status

{st.session_state.stage.upper()}

""", unsafe_allow_html=True) angle_box.markdown(f"""

Knee Angle

{int(angle)}°

""", unsafe_allow_html=True) progress_bar.progress(st.session_state.reps / 10) time.sleep(0.03) cap.release() # --- SUCCESS --- if st.session_state.reps >= 10: st.balloons() st.success("🎯 Goal Reached! Workout Complete.")
