import streamlit as st import cv2 import mediapipe as mp import numpy as np from firebase_admin import firestore # --- STYLING --- st.set_page_config(page_title="FitRise AI", layout="centered") st.markdown(""" """, unsafe_content_allowed=True) # --- MEDIAPIPE INITIALIZATION --- mp_drawing = mp.solutions.drawing_utils mp_pose = mp.solutions.pose def calculate_angle(a, b, c): a = np.array(a) b = np.array(b) c = np.array(c) radians = np.arctan2(c[1]-b[1], c[0]-b[0]) - np.arctan2(a[1]-b[1], a[0]-b[0]) angle = np.abs(radians*180.0/np.pi) if angle > 180.0: angle = 360-angle return angle # --- APP INTERFACE --- st.title("Squat Protocol Active") col1, col2 = st.columns([2, 1]) with col2: st.markdown("
", unsafe_content_allowed=True) rep_display = st.empty() stage_display = st.empty() st.markdown("
", unsafe_content_allowed=True) progress_bar = st.progress(0) with col1: frame_placeholder = st.empty() # --- INITIALIZE STATE --- if 'reps' not in st.session_state: st.session_state.reps = 0 if 'stage' not in st.session_state: st.session_state.stage = "up" # --- CAMERA LOOP --- cap = cv2.VideoCapture(0) # Higher confidence prevents "flickering" reps with mp_pose.Pose(min_detection_confidence=0.7, min_tracking_confidence=0.7) as pose: while cap.isOpened() and st.session_state.reps < 10: ret, frame = cap.read() if not ret: st.error("Camera not found.") break # 1. Prepare Image # Flip for mirror effect, convert to RGB for MediaPipe image = cv2.cvtColor(cv2.flip(frame, 1), cv2.COLOR_BGR2RGB) image.flags.writeable = False # 2. Detect Pose results = pose.process(image) # 3. Draw & Logic image.flags.writeable = True image = cv2.cvtColor(image, cv2.COLOR_RGB2BGR) try: landmarks = results.pose_landmarks.landmark # Get coordinates for Hip, Knee, Ankle (Right Side) hip = [landmarks[mp_pose.PoseLandmark.RIGHT_HIP.value].x, landmarks[mp_pose.PoseLandmark.RIGHT_HIP.value].y] knee = [landmarks[mp_pose.PoseLandmark.RIGHT_KNEE.value].x, landmarks[mp_pose.PoseLandmark.RIGHT_KNEE.value].y] ankle = [landmarks[mp_pose.PoseLandmark.RIGHT_ANKLE.value].x, landmarks[mp_pose.PoseLandmark.RIGHT_ANKLE.value].y] angle = calculate_angle(hip, knee, ankle) # VISUAL FEEDBACK: Draw the angle on the knee # Convert normalized to pixel for text placement h, w, _ = image.shape knee_pos = tuple(np.multiply(knee, [w, h]).astype(int)) cv2.putText(image, str(int(angle)), knee_pos, cv2.FONT_HERSHEY_SIMPLEX, 0.5, (0, 255, 136), 2, cv2.LINE_AA) # SQUAT COUNTER LOGIC # Standing = ~170 degrees | Squatting = < 100 degrees if angle > 160: st.session_state.stage = "up" if angle < 100 and st.session_state.stage == "up": st.session_state.stage = "down" st.session_state.reps += 1 # Optional: Add a sound trigger here if you want it to beep except Exception as e: pass # Pose not fully in frame # 4. Render Layout mp_drawing.draw_landmarks(image, results.pose_landmarks, mp_pose.POSE_CONNECTIONS, mp_drawing.DrawingSpec(color=(0, 255, 136), thickness=2, circle_radius=2), mp_drawing.DrawingSpec(color=(255, 255, 255), thickness=2, circle_radius=2)) frame_placeholder.image(image, channels="BGR", use_column_width=True) # Update text displays rep_display.markdown(f"## REPS: {st.session_state.reps}") stage_display.markdown(f"**STATUS:** {st.session_state.stage.upper()}") progress_bar.progress(st.session_state.reps / 10) cap.release() # --- SUCCESS STATE --- if st.session_state.reps >= 10: st.balloons() st.success("Target Reached. Alarm Deactivated.") # Your Firebase update logic here
